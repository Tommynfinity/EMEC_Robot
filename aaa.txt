        public SerialPortController(string port, int baudRate)
        {
            portName = port;
            serialPort = new SerialPort(port, baudRate, Parity.None, 8, StopBits.One);
            serialPort.Encoding = Encoding.UTF8;
            serialPort.DataReceived += new SerialDataReceivedEventHandler(DataReceivedHandler);
        }


        private volatile bool controlloTimeoutInCorso = false;

        public bool ControlloTimeoutInCorso => controlloTimeoutInCorso;

        public void StopControlloTimeout()
        {
            controlloTimeoutInCorso = false;
            Console.WriteLine($"⏹️ Controllo timeout fermato su {portName}");
        }

        public async void AvviaControlloTimeout()
        {

            if (controlloTimeoutInCorso) return; // evita più avvii

            controlloTimeoutInCorso = true;
            _ = Task.Run(async () =>
            {
                while (controlloTimeoutInCorso)
                {
                    sequencer.ControllaTimeoutStep(SendJson, GestisciFallimentoSessione);
                    await Task.Delay(500);
                   
                }
            });
        }


        private void GestisciFallimentoSessione()
        {
            controlloTimeoutInCorso = false;
            Console.WriteLine($"Sessione fallita su {portName}. Chiudo.");

            var mainForm = Application.OpenForms["Dashboard"] as Dashboard;
            TimeSpan time = DateTime.Now.TimeOfDay;
            mainForm?.MostraErrore($"{time.Hours}:{time.Minutes}:{time.Seconds} -  ERR_NO_RESPONSE: {portName} .");
            // 1. Salva log su DB con risultato "Fallito"
            // 2. Reset sequencer o segna stato basetta come "errore"
        }        
        
        public void SerialFlush()
        {
            if (serialPort.IsOpen)
            {
                try
                {
                    serialPort.DiscardInBuffer();
                    serialPort.DiscardOutBuffer();
                    while (serialPort.BytesToRead > 0)
                    {
                        serialPort.ReadExisting();
                    }
                    Console.WriteLine("Seriale pulita");
                }
                catch (Exception ex)
                {
                    Console.WriteLine($"❌ Errore nel flush seriale: {ex.Message}");
                }
            }
        }

        public bool Start()
        {
            try
            {
                if (!serialPort.IsOpen)
                    serialPort.Open();
                serialPort.DiscardOutBuffer();
                return true;
            }
            catch (Exception ex)
            {
                var mainForm = Application.OpenForms["Dashboard"] as Dashboard;
                TimeSpan time = DateTime.Now.TimeOfDay;
                mainForm?.MostraErrore($"{time.Hours}:{time.Minutes}:{time.Seconds} - ERR_COM_NOT_FOUND: {portName}");
                return false;
            }
        }

        public void Stop()
        {
            if (serialPort.IsOpen)
                serialPort.Close();
        }

        public GestoreSequenza GetSequencer() => sequencer;

        public void SendJson(Package pack)
        {
            if (!controlloTimeoutInCorso)
            {
                Console.WriteLine($"⛔ Invio bloccato su {portName}: controllo timeout non attivo.");
                return;
            }

            gestore.AggiungiRichiesta(pack);
            string json = "\x02" + JsonConvert.SerializeObject(pack) + "\x03";
            Console.WriteLine($"📤 [{portName}] Inviato: {json}");
            serialPort.WriteLine(json);
            Thread.Sleep(100);
            var step = sequencer.GetStepCorrente();
            if (step != null && step.PacchettoDaInviare == null)
                step.PacchettoDaInviare = pack;
        }


        private void DataReceivedHandler(object sender, SerialDataReceivedEventArgs e)

            try
            {
                string data = serialPort.ReadExisting();
                buffer.Append(data);

                while (buffer.ToString().Contains("\n"))
                {
                    string fullLine = buffer.ToString();

                    string jsonLine = fullLine.Substring(0, index).Trim();
                    buffer.Remove(0, index + 1);

                    try
                    {
                        var pacchetto = JsonConvert.DeserializeObject<Package>(jsonLine);

                        var step = sequencer.GetStepCorrente();
                        if (step != null &&
                            step.TipoRispostaAttesa == pacchetto.StatoConversazione &&
                            (step.ValidatoreRisposta == null || step.ValidatoreRisposta(pacchetto)))
                        {
                            step.PacchettoRicevuto = pacchetto;
                            step.Completata = true;
                            Console.WriteLine($"✅ [{portName}] Step completato: {step.Nome}");
                            sequencer.PassaAlProssimo();
                        }

                        var mainForm = Application.OpenForms["Dashboard"] as Dashboard;

                        switch (pacchetto.StatoConversazione)
                        {
                            case "Request":
                                Console.WriteLine($"🔁 [{portName}] Richiesta ricevuta.");
                                if (!string.IsNullOrWhiteSpace(pacchetto.Comando))
                                {
                                    if (pacchetto.Comando == "RichiediStato")
                                    {
                                        var risposta = new Package
                                        {
                                            Comando = "RichiediStato",
                                            Token = pacchetto.Token,
                                            Data = DateTime.Now,
                                            Messaggio = "Stato: OK",
                                            StatoConversazione = "Response"
                                        };
                                        string rispostaJson = JsonConvert.SerializeObject(risposta);
                                        serialPort.WriteLine(rispostaJson);
                                        Console.WriteLine($"📤 [{portName}] Inviata risposta: {rispostaJson}");
                                    }
                                    else
                                    {
                                        Console.WriteLine($"⚠️ [{portName}] Comando sconosciuto: {pacchetto.Comando}");
                                    }
                                }
                                break;

                            case "Response":
                                Console.WriteLine($"📩 [{portName}] Risposta ricevuta.");

                                if (!string.IsNullOrWhiteSpace(pacchetto.Token) && gestore.ContieneRichiesta(pacchetto.Token))
                                {
                                    var richiestaOriginale = gestore.RimuoviRichiesta(pacchetto.Token);
                                    Console.WriteLine($"🔗 [{portName}] Risposta correlata a: {richiestaOriginale.Comando}");

                                                    case "VCL":
                                                        RobotController.Instance.WriteRegister(Registers.VCL_Start, 1);

        public void ScriviDopoDelay(ushort registro, ushort valore, int delayMs)
        {
            _ = Task.Run(async () =>
            {
                await Task.Delay(delayMs);
                RobotController.Instance.WriteRegister(registro, valore);
                Console.WriteLine($"⏱️ Registro {registro} scritto con {valore} dopo {delayMs}ms");
            });
        }

        private void SaveRecordOnDB(SessioneTest sessione, RisultatiTest risultato)
        {
            using (var connection = new MySqlConnection(DB.ConnectionString))
            {
                connection.Open();

                string insertSessionQuery = @"
            INSERT INTO sessione_test
            (identification, id_basetta, vassoio_deposito, vassoio_prelievo,
             pezzi_da_collaudare, pezzi_ok, pezzi_ko, completata)
            VALUES (@identification, @id_basetta, @vassoio_deposito, @vassoio_prelievo,
                    @pezzi_da_collaudare, @pezzi_ok, @pezzi_ko, @completata);
            SELECT LAST_INSERT_ID();";

                int sessioneId;
                using (var cmd = new MySqlCommand(insertSessionQuery, connection))
                {
                    cmd.Parameters.AddWithValue("@identification", sessione.Identification);
                    cmd.Parameters.AddWithValue("@id_basetta", sessione.IdBasetta);
                    cmd.Parameters.AddWithValue("@vassoio_deposito", sessione.VassoioDeposito);
                    cmd.Parameters.AddWithValue("@vassoio_prelievo", sessione.VassoioPrelievo);
                    cmd.Parameters.AddWithValue("@pezzi_da_collaudare", sessione.PezziDaCollaudare);
                    cmd.Parameters.AddWithValue("@pezzi_ok", sessione.PezziOK);
                    cmd.Parameters.AddWithValue("@pezzi_ko", sessione.PezziKO);
                    cmd.Parameters.AddWithValue("@completata", sessione.Completata);
                    sessioneId = Convert.ToInt32(cmd.ExecuteScalar());
                }

                string insertResultQuery = @"
            INSERT INTO risultati_test
            (sessione_id, coil_ton, freq, led, gate_ton, esito, dut_code, indirizzo)
            VALUES (@sessione_id, @coil_ton, @freq, @led, @gate_ton, @esito, @dut_code, @indirizzo)";

                using (var cmd = new MySqlCommand(insertResultQuery, connection))
                {
                    cmd.Parameters.AddWithValue("@sessione_id", sessioneId);
                    cmd.Parameters.AddWithValue("@coil_ton", risultato.CoilTon);
                    cmd.Parameters.AddWithValue("@freq", risultato.Freq);
                    cmd.Parameters.AddWithValue("@led", risultato.Led);
                    cmd.Parameters.AddWithValue("@gate_ton", risultato.GateTon);
                    cmd.Parameters.AddWithValue("@esito", risultato.Esito);
                    cmd.Parameters.AddWithValue("@dut_code", risultato.DutCode);
                    cmd.Parameters.AddWithValue("@indirizzo", risultato.Indirizzo);
                    cmd.ExecuteNonQuery();
                }
            }
        }


        private ushort _MappingCodsc(string cod)
        {
            using (MySqlConnection connection = new MySqlConnection(DB.ConnectionString))
            {
                try
                {
                    connection.Open();

                    string query = $"SELECT map FROM config WHERE Codice_scheda = '{cod}';";
                    using (MySqlCommand command = new MySqlCommand(query, connection))
                    {
                        MySqlDataReader reader = command.ExecuteReader();

                        while (reader.Read())
                        {
                            return Convert.ToUInt16(reader["map"]);
                        }
                    }
                }
                catch (MySqlException mysqlEx)
                {

                }
                catch (Exception ex)
                {
                    MessageBox.Show($"Errore: {ex}");
                }
                connection.Close();
            }
            return 0;
        }
    }
